static std::string readAll(ZzPty& pty)
{
    std::string out;
    std::byte buf[4096];
#if defined(_WIN32)
    const DWORD deadline = ::GetTickCount() + 5000;
    for (;;) {
        const std::ptrdiff_t n = pty.read(std::span<std::byte>(buf, sizeof(buf)));
        if (n > 0) { out.append(reinterpret_cast<const char*>(buf), (std::size_t)n); continue; }

    // env 为空时 nullptr 走 execvp 纯继承（与旧行为逐字节一致）。

        for (char** e = environ; *e != nullptr; ++e)
            mergedEnv.emplace_back(*e);
        for (const std::string& kv : cfg.env) {
            const std::string key = kv.substr(0, kv.find('='));
            std::erase_if(mergedEnv, [&](const std::string& e) {

                       && e[key.size()] == '=';
            });
            mergedEnv.push_back(kv); // 后写覆盖先写（同名已删）
        }
        envp.reserve(mergedEnv.size() + 1);
        for (std::string& e : mergedEnv)
            envp.push_back(e.data());
        envp.push_back(nullptr);

```cpp
// M18：合并环境块（双 NUL 结尾宽字符串）：父环境 + cfg.env 覆盖同名。
// Windows 环境块约定按键（大写不敏感）字母序排列。
struct ZzCaseInsensitiveLess {
    bool operator()(const std::wstring& a, const std::wstring& b) const
    {
        return ::CompareStringOrdinal(a.c_str(), -1, b.c_str(), -1, TRUE) == CSTR_LESS_THAN;
    }
};

std::vector<wchar_t> buildEnvBlock(const std::vector<std::string>& extra)
{
    std::map<std::wstring, std::wstring, ZzCaseInsensitiveLess> merged;
    if (LPWCH block = ::GetEnvironmentStringsW()) {
        for (LPCWSTR p = block; *p != L'\0'; p += std::wcslen(p) + 1) {
            const std::wstring entry(p);
            const auto eq = entry.find(L'=');
            if (eq != std::wstring::npos)
                merged[entry.substr(0, eq)] = entry.substr(eq + 1);
        }
        ::FreeEnvironmentStringsW(block);
    }
    for (const std::string& kv : extra) {
        const auto eq = kv.find('=');
        if (eq == std::string::npos)
            continue; // 非法条目（无 =）跳过
        merged[toWide(kv.substr(0, eq))] = toWide(kv.substr(eq + 1)); // 同键覆盖
    }
    std::vector<wchar_t> out;
    for (const auto& [key, value] : merged) {
        out.insert(out.end(), key.begin(), key.end());
        out.push_back(L'=');
        out.insert(out.end(), value.begin(), value.end());
        out.push_back(L'\0');
    }
    out.push_back(L'\0'); // 终止双 NUL
    return out;
}
```
